# =============================================================================
# GTI MS Teams Bot (Agentic) — Azure Container Apps Dependencies
# =============================================================================

# ASGI web framework and server — see main.py. Runs the exact same app/
# business logic that azure/azure-bot-function runs, just without the Azure
# Functions host: no platform-imposed HTTP response ceiling (Functions caps
# every plan at 230s regardless of host.json; a container has none), which
# is the whole reason this variant exists.
fastapi==0.141.1
uvicorn[standard]==0.52.4

# Async HTTP client for everything that talks to an external API: the GTI
# Agentic API, Microsoft Graph, and the Bot Framework Connector API itself
# (sending/editing/deleting Teams messages, and the bot's own OAuth token) —
# see app/gti/client.py, app/graph/client.py, app/teams/bot_client.py,
# app/teams/attachments.py. Every call is awaited natively; nothing here
# runs in a worker thread.
httpx==0.28.1

# Async transport for azure-identity/.aio, azure-storage-blob/.aio, and
# azure-data-tables/.aio below — required by those packages' aio submodules
# even though nothing here calls aiohttp directly.
aiohttp==3.14.3

# User-Assigned Managed Identity authentication (production) for the bot's
# own Bot Framework Connector token and Microsoft Graph token — see
# app/teams/bot_client.py, app/graph/client.py (both use azure.identity.aio).
# Works identically on Container Apps as it did on Functions: attach the
# same User-Assigned Managed Identity to the Container App resource.
azure-identity==1.25.3

# Type-safe, validated configuration from environment variables.
pydantic-settings==2.15.0

# Local development only — load a .env file into the environment when
# running `uvicorn main:app --reload` directly.
python-dotenv==1.2.3

# Persists the configured output-format instructions as a JSON blob in an
# Azure Storage account (app/output_format_store.py, via azure.storage.blob.aio).
azure-storage-blob==12.30.1

# Persists per-thread GTI session IDs as Table Storage entities in the same
# storage account (app/gti/session_store.py, via azure.data.tables.aio) —
# atomic per-row upserts avoid the read-modify-write race a shared JSON blob
# would have.
azure-data-tables==12.7.0

# JWT decoding and key verification for Bot Framework security validation
# (app/teams/auth.py) — [crypto] pulls in `cryptography`, required for RS256.
# This check is pure CPU (no I/O), so it stays synchronous even though it
# runs inside an `async def` route in main.py — it's fast enough not to
# matter, and PyJWT has no async API to call instead.
pyjwt[crypto]==2.14.0
