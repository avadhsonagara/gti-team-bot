# =============================================================================
# GTI MS Teams Bot (Agentic) — Azure Functions Dependencies
# =============================================================================

# Azure Functions Python worker (v2 programming model) — see function_app.py.
azure-functions>=1.21.0

# HTTP client for everything that talks to an external API: the GTI Agentic
# API, Microsoft Graph, and the Bot Framework Connector API itself (sending/
# editing/deleting Teams messages, and the bot's own OAuth token) — see
# app/gti/client.py, app/graph/client.py, app/teams/bot_client.py,
# app/teams/attachments.py. Plain synchronous calls throughout — no
# microsoft-teams-apps SDK, no async, matching the style already used in
# azure/rs-alerts.
requests>=2.31.0

# User-Assigned Managed Identity authentication (production) for the bot's
# own Bot Framework Connector token and Microsoft Graph token — see
# app/teams/bot_client.py, app/graph/client.py. Declared explicitly here:
# the microsoft-teams-apps SDK previously pulled this in transitively: now
# that it's gone, this dependency must be listed directly.
azure-identity>=1.17.0

# Type-safe, validated configuration from environment variables (Azure
# Function App Application Settings are exposed as env vars at runtime).
pydantic-settings>=2.0

# Local development only — load .env file into environment when running a
# script directly outside of `func start` (which reads local.settings.json
# instead).
python-dotenv>=1.0

# Persists the configured output-format instructions as a JSON blob in the
# Function App's own storage account (app/output_format_store.py).
azure-storage-blob>=12.19.0

# Persists per-thread GTI session IDs as Table Storage entities in the same
# storage account (app/gti/session_store.py) — atomic per-row upserts avoid
# the read-modify-write race a shared JSON blob would have.
azure-data-tables>=12.5.0

# JWT decoding and key verification for Bot Framework security validation
# (app/teams/auth.py) — [crypto] pulls in `cryptography`, required for RS256.
# Also declared explicitly for the same reason as azure-identity above.
pyjwt[crypto]>=2.8.0
